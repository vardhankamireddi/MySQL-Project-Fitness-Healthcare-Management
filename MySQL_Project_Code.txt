    -> BEGIN
    ->     DECLARE v_id INT;
    ->     DECLARE v_name VARCHAR(30);
    ->     DECLARE v_cal DECIMAL(8,1);
    ->     DECLARE v_done INT DEFAULT 0;
    ->     DECLARE cur CURSOR FOR SELECT user_id, first_name FROM users WHERE status = 'Active';
    ->     DECLARE CONTINUE HANDLER FOR NOT FOUND SET v_done = 1;
    ->
    ->     DROP TEMPORARY TABLE IF EXISTS tmp_report;
    ->     CREATE TEMPORARY TABLE tmp_report (user_id INT, first_name VARCHAR(30), calories_burned DECIMAL(8,1));
    ->
    ->     OPEN cur;
    ->     read_loop: LOOP
    ->         FETCH cur INTO v_id, v_name;
    ->         IF v_done = 1 THEN LEAVE read_loop; END IF;
    ->         SELECT IFNULL(SUM(calories_burned), 0) INTO v_cal FROM v_workout_calories WHERE user_id = v_id;
    ->         INSERT INTO tmp_report VALUES (v_id, v_name, v_cal);
    ->     END LOOP;
    ->     CLOSE cur;
    ->
    ->     SELECT * FROM tmp_report ORDER BY calories_burned DESC;
    -> END$$
